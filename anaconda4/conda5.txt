
# ==========================================
# PRACTICAL 5: JENKINS + GITHUB ON UBUNTU
# ==========================================

# 1. Update Ubuntu packages
sudo apt update
sudo apt upgrade -y

# 2. Install required utilities
sudo apt install -y wget curl git fontconfig

# 3. Install Java JDK 21
sudo apt install -y openjdk-21-jdk

# 4. Verify Java installation
java -version
javac -version

# 5. Configure JAVA_HOME
JAVA_PATH=$(dirname "$(dirname "$(readlink -f "$(which java)")")")
echo "export JAVA_HOME=$JAVA_PATH" >> ~/.bashrc
echo 'export PATH=$JAVA_HOME/bin:$PATH' >> ~/.bashrc
source ~/.bashrc

# 6. Verify JAVA_HOME
echo $JAVA_HOME

# 7. Add Jenkins repository signing key
sudo mkdir -p /etc/apt/keyrings
sudo wget -O /etc/apt/keyrings/jenkins-keyring.asc \
https://pkg.jenkins.io/debian-stable/jenkins.io-2026.key

# 8. Add Jenkins LTS repository
echo "deb [signed-by=/etc/apt/keyrings/jenkins-keyring.asc] \
https://pkg.jenkins.io/debian-stable binary/" | \
sudo tee /etc/apt/sources.list.d/jenkins.list > /dev/null

# 9. Update package list
sudo apt update

# 10. Install Jenkins
sudo apt install -y jenkins

# 11. Enable and start Jenkins
sudo systemctl enable jenkins
sudo systemctl start jenkins

# 12. Check Jenkins status
sudo systemctl status jenkins

# 13. Get initial administrator password
sudo cat /var/lib/jenkins/secrets/initialAdminPassword

# 14. Check Git version
git --version
